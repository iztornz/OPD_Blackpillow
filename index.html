ตั้งค่าเชื่อมต่อ Google Sheets (แท็บ raw data)

โค้ดสำหรับวางใน Google Sheet > ส่วนขยาย > Apps Script: 
const SPREADSHEET_ID = '1Pm0-pxYRgFVvX_iZvKd663FdaeT93guYwvMIebxVl3s';
const TARGET_TAB_NAME = 'raw data';

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const payload = JSON.parse(e.postData.contents);
    const rows = Array.isArray(payload.rows) ? payload.rows : [];
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let sheet = ss.getSheetByName(TARGET_TAB_NAME) || ss.insertSheet(TARGET_TAB_NAME);

    if (sheet.getLastRow() === 0) {
      sheet.getRange(1, 1, 1, 9).setValues([[
        'ชื่อ', 'นามสกุล', 'อายุ', 'เพศ', 'สัญชาติ',
        'วันเดือนปีเกิด', 'เบอร์โทรศัพท์', 'วันเดือนปีที่เข้ารับบริการ', 'รายการที่มาทำ'
      ]]).setFontWeight('bold');
    }

    if (rows.length > 0) {
      const startRow = sheet.getLastRow() + 1;
      const values = rows.map(r => [
        r.firstName || '', r.lastName || '', r.age || '', r.gender || '',
        r.nationality || '', r.dob || '', r.phone || '', r.serviceDate || '', r.treatment || ''
      ]);
      sheet.getRange(startRow, 6, values.length, 3).setNumberFormat('@');
      sheet.getRange(startRow, 1, values.length, 9).setValues(values);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'ok', count: rows.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

วิธีติดตั้ง: เปิดไฟล์ Google Sheet → ส่วนขยาย > Apps Script → วางโค้ดนี้ → กด การทำให้ใช้งานได้ (Deploy) > รายการใหม่ > เว็บแอป (ผู้มีสิทธิ์เข้าถึง: ทุกคน/Anyone) → คัดลอก URL มาวางในช่องด้านบน
