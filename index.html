  5. คัดลอก Web App URL มาวางในช่องด้านบน เป็นอันเสร็จสมบูรณ์!

โค้ด Google Apps Script (Code.gs) สำหรับแท็บ raw data

ผูกกับ Spreadsheet ID 1Pm0-pxYRgFVvX_iZvKd663FdaeT93guYwvMIebxVl3s และแท็บ raw data โดยตรง

/**
 * Blackpillow OPD Batch OCR -> Auto-Sync ลงแท็บ "raw data"
 * Spreadsheet: https://docs.google.com/spreadsheets/d/1Pm0-pxYRgFVvX_iZvKd663FdaeT93guYwvMIebxVl3s/edit
 */
const SPREADSHEET_ID = '1Pm0-pxYRgFVvX_iZvKd663FdaeT93guYwvMIebxVl3s';
const TARGET_TAB_NAME = 'raw data';

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);

  try {
    const rawBody = e.postData ? e.postData.contents : '{}';
    const payload = JSON.parse(rawBody);
    const rows = Array.isArray(payload.rows) ? payload.rows : [payload];

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    let sheet = ss.getSheetByName(TARGET_TAB_NAME);
    if (!sheet) {
      sheet = ss.insertSheet(TARGET_TAB_NAME);
    }

    // สร้างหัวตาราง 9 คอลัมน์อัตโนมัติหากแท็บ raw data ยังว่างอยู่
    if (sheet.getLastRow() === 0) {
      const headers = [
        'ชื่อ',
        'นามสกุล',
        'อายุ',
        'เพศ',
        'สัญชาติ',
        'วันเดือนปีเกิด',
        'เบอร์โทรศัพท์',
        'วันเดือนปีที่เข้ารับบริการ',
        'รายการที่มาทำ'
      ];
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight('bold');
    }

    if (rows.length > 0) {
      const startRow = sheet.getLastRow() + 1;
      const valuesToAppend = rows.map(item => {
        // บังคับเบอร์โทรศัพท์เป็น 0xxxxxxxxx (10 หลักติดกัน)
        let cleanPhone = String(item.phone || '').replace(/\D/g, '');
        if (cleanPhone.length === 9 && !cleanPhone.startsWith('0')) {
          cleanPhone = '0' + cleanPhone;
        }
        return [
          String(item.firstName || '').trim(),
          String(item.lastName || '').trim(),
          String(item.age || '').trim(),
          String(item.gender || '').trim(),
          String(item.nationality || 'ไทย').trim(),
          String(item.birthDate || '').trim(),
          cleanPhone,
          String(item.serviceDate || '').trim(),
          String(item.treatments || '').trim()
        ];
      });

      const targetRange = sheet.getRange(startRow, 1, valuesToAppend.length, 9);
      // ตั้งค่า Format คอลัมน์ วันเกิด (Col 6), เบอร์โทร (Col 7), วันรับบริการ (Col 8) เป็น Plain Text (@) เพื่อไม่ให้เลข 0 นำหน้าหาย
      sheet.getRange(startRow, 6, valuesToAppend.length, 3).setNumberFormat('@');
      targetRange.setValues(valuesToAppend);
    }

    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      insertedRows: rows.length,
      sheetName: TARGET_TAB_NAME
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

สรุปกฎการจัดรูปแบบทั้ง 9 ฟิลด์ก่อนส่งเข้า raw data

  • 1. ชื่อ: ตัดคำนำหน้า (K., คุณ, นาย, นางสาว, น.ส.) ออกทั้งหมด
  • 2. นามสกุล: หากชื่อและนามสกุลเขียนต่อกันในช่องเดียว ระบบจะแยกคำหลังเว้นวรรคมาใส่ช่องนามสกุลให้อัตโนมัติ
  • 3–5. อายุ / เพศ / สัญชาติ: อายุเป็นตัวเลขจำนวนเต็ม, เพศจัดรูปแบบเป็น ญ หรือ ช, สัญชาติค่าเริ่มต้น ไทย
  • 6 & 8. วันเกิด & วันรับบริการ: แปลงเป็น dd/mm/yyyy (พ.ศ.) เสมอ เช่น 30 กค 2550 → 30/07/2550, 30/8/69 → 30/08/2569
  • 7. เบอร์โทรศัพท์: ล็อกเป็น Text 10 หลักติดกัน 0xxxxxxxxx ไม่ให้ Google Sheets ตัดเลข 0 ตัวหน้าออก
  • 9. รายการที่มาทำ: สกัดรายการหัตถการทั้งหมดจากใบ OPD ไม่ว่าจะเขียนอยู่ในช่อง Treatment หรือในตาราง
